import Foundation
@testable import GameCore
@testable import GamePresentation
import XCTest

// tools/map3d-preview: copy into Tests/GamePresentationTests/ to use, and
// delete it again before committing (it writes files and is not a test).
// Writes a map's 3D scene for the WebGL preview (README.md beside it).
final class Map3DPreviewDump: XCTestCase {
    func testDump() throws {
        guard let out = ProcessInfo.processInfo.environment["MAP3D_DUMP"] else { throw XCTSkip("no MAP3D_DUMP") }
        let world: GameWorld
        let root = URL(fileURLWithPath: #filePath).deletingLastPathComponent().deletingLastPathComponent().deletingLastPathComponent()
            .appendingPathComponent("RailwayGameApp/Resources/")
        func resource(_ path: String) throws -> Data { try Data(contentsOf: root.appendingPathComponent(path)) }
        switch ProcessInfo.processInfo.environment["MAP3D_WORLD"] {
        case "towns":
            let towns = Land.townCentres(seed: 1, in: GameWorld.newGameBounds)
            world = try newGameLine(through: [towns[0], towns[1]])
        case "taipei":
            world = GameWorld.newGame(anchor: GeoAnchor(latitudeDegrees: 25.0478, longitudeDegrees: 121.5170)!)
        // Decision 239: the city tiles' own origins.
        case "kaohsiung":
            world = GameWorld.newGame(anchor: GeoAnchor(latitudeDegrees: 22.6394, longitudeDegrees: 120.3024)!)
        case "taichung":
            world = GameWorld.newGame(anchor: GeoAnchor(latitudeDegrees: 24.1371, longitudeDegrees: 120.6869)!)
        case "realdemo":
            let railways = try RealRailways(
                lines: resource("RealRailways/track_lines.geojson"), stations: resource("RealRailways/track_stations.geojson"),
                names: resource("RealRailways/station_names.json"),
                pingxiRuns: RealLineRuns(data: resource("RealRailways/tra_pingxi_runs.json")))
            let heights = try HeightGrid(data: resource("RealWorld/taiwan_heights.dat"))
            let water = try WaterGrid(data: resource("RealWorld/taiwan_water.json"))
            let frame = RealWorldFrame(anchor: RealWorldDemo.anchor, bounds: GameWorld.newGameBounds)
            world = RealWorldDemo.make(in: .traditionalChinese, railways: railways, water: water.cells(frame: frame, bounds: GameWorld.newGameBounds),
                                       steep: water.steepCells(frame: frame, bounds: GameWorld.newGameBounds), heights: heights)
        default:
            world = DemoWorld.make(in: .english)
        }
        let tiles = ProcessInfo.processInfo.environment["MAP3D_BASEMAP"] == "1"
            ? try BaseMapTiles(data: Data(contentsOf: root.appendingPathComponent("BaseMap/taiwan.pmtiles"), options: .alwaysMapped)) : nil
        let cityTiles = ProcessInfo.processInfo.environment["MAP3D_CITY"] == "1"
            ? try ["kaohsiung", "taichung"].map { area in
                try CityTileArea(manifest: resource("BaseMap/city/\(area)/manifest.json")) { try? resource("BaseMap/city/\(area)/\($0)") }
            } : []
        // MAP3D_HEIGHTS=1: the heights file's ground where the world has
        // not read its own, as the app passes it.
        let heights = ProcessInfo.processInfo.environment["MAP3D_HEIGHTS"] == "1"
            ? try HeightGrid(data: resource("RealWorld/taiwan_heights.dat")) : nil
        let scene = Map3DScene(world: world, baseMap: tiles, cityTiles: cityTiles, heights: heights)
        let crop = ProcessInfo.processInfo.environment["MAP3D_CROP"].map { $0.split(separator: ":").map { Float(String($0))! } }
        let chunks = scene.chunks.filter { chunk in
            guard let crop else { return true }
            return chunk.maximum.x > crop[0] - crop[2] && chunk.minimum.x < crop[0] + crop[2]
                && chunk.maximum.z > crop[1] - crop[2] && chunk.minimum.z < crop[1] + crop[2]
        }
        var floats: [Float] = []
        var indices: [UInt32] = []
        // MAP3D_DETAIL=far: each chunk as the app draws it from afar
        // (decision 238): its ground and track, then its city simplified.
        let detailFar = ProcessInfo.processInfo.environment["MAP3D_DETAIL"] == "far"
        // An instance's model moved into place (the app draws them as
        // instances).
        func expand(_ prop: Map3DInstance) {
            let model = Map3DModelPack.standard.models[prop.model], t = prop.transform
            let first = UInt32(floats.count / 12)
            for v in model.vertices {
                let p = Map3DPoint(t[0] * v.x + t[4] * v.y + t[8] * v.z + t[12], t[1] * v.x + t[5] * v.y + t[9] * v.z + t[13],
                                   t[2] * v.x + t[6] * v.y + t[10] * v.z + t[14])
                let n = (Map3DPoint(t[0] * v.nx + t[4] * v.ny + t[8] * v.nz, t[1] * v.nx + t[5] * v.ny + t[9] * v.nz,
                                    t[2] * v.nx + t[6] * v.ny + t[10] * v.nz).normalized) ?? Map3DPoint(0, 1, 0)
                let c = v.color, share = Float(c >> 24) / 255
                func mix(_ own: UInt32, _ tint: Float) -> Float { tint + (Float(own) / 255 - tint) * share }
                floats += [p.x, p.y, p.z, n.x, n.y, n.z, mix(c & 0xFF, prop.stripe.0), mix((c >> 8) & 0xFF, prop.stripe.1),
                           mix((c >> 16) & 0xFF, prop.stripe.2), 0, 0, v.glows ? Map3DVertex.glowing : -1]
            }
            indices += (0..<UInt32(model.vertices.count)).map { $0 + first }
        }
        for chunk in chunks {
            // The vertices of both; the indices of the base and of the
            // simplified city (afar), or all of the full mesh (near).
            let vertices = detailFar ? chunk.mesh.vertices + chunk.far.vertices : chunk.mesh.vertices
            let chunkIndices = detailFar
                ? Array(chunk.mesh.indices.prefix(chunk.baseIndexCount)) + chunk.far.indices.map { $0 + UInt32(chunk.mesh.vertices.count) }
                : chunk.mesh.indices
            let base = UInt32(floats.count / 12)
            for v in vertices {
                let c = v.color
                floats += [v.x, v.y, v.z, v.nx, v.ny, v.nz,
                           Float(c & 0xFF) / 255, Float((c >> 8) & 0xFF) / 255, Float((c >> 16) & 0xFF) / 255, v.u, v.v, v.layer]
            }
            indices += chunkIndices.map { $0 + base }
            // Decision 244: the street furniture, near, each instance's
            // model moved into place (the app draws them as instances).
            guard !detailFar else { continue }
            for prop in chunk.props { expand(prop) }
        }
        // Decision 259: MAP3D_LIFE=x:z:distance, the people and vehicles a
        // camera that far from that target draws at 10:00:00 of a clock,
        // each moved into place; MAP3D_BUSY=1 as if every station were busy.
        if let life = ProcessInfo.processInfo.environment["MAP3D_LIFE"].map({ $0.split(separator: ":").map { Float(String($0))! } }) {
            var activity = Map3DStreetLife.Activity(world: world)
            if ProcessInfo.processInfo.environment["MAP3D_BUSY"] == "1" {
                for spot in scene.streetLife.spots {
                    guard let station = spot.station else { continue }
                    activity.platform[station] = Map3DStreetLife.platformCrowd(waiting: 400, platforms: 2)
                    activity.front[station] = Map3DStreetLife.frontCrowd(trips: 8_000)
                }
            }
            let frame = scene.streetLife.frame(target: Map3DPoint(life[0], 0, life[1]), distance: life[2], time: 36_000,
                                               detail: ProcessInfo.processInfo.environment["MAP3D_LIFE_DETAIL"] == "reduced" ? .reduced : .full,
                                               activity: activity)
            print("MAP3D_LIFE people \(frame.people) vehicles \(frame.vehicles) models \(Set(frame.instances.map(\.model)).count)")
            frame.instances.forEach(expand)
        }
        var shadowFloats: [Float] = [], shadowIndices: [UInt32] = []
        for chunk in chunks {
            let base = shadowFloats.count / 7
            // Position, then the colour (black for a shadow, a light's
            // pool otherwise) and its alpha.
            for v in chunk.shadows.vertices {
                shadowFloats += [v.x, v.y, v.z, Float(v.color & 0xFF) / 255, Float((v.color >> 8) & 0xFF) / 255,
                                 Float((v.color >> 16) & 0xFF) / 255, Float(v.color >> 24) / 255]
            }
            shadowIndices += chunk.shadows.indices.map { $0 + UInt32(base) }
        }
        try shadowFloats.withUnsafeBytes { try Data($0).write(to: URL(fileURLWithPath: out + ".svertices")) }
        try shadowIndices.withUnsafeBytes { try Data($0).write(to: URL(fileURLWithPath: out + ".sindices")) }
        try floats.withUnsafeBytes { try Data($0).write(to: URL(fileURLWithPath: out + ".vertices")) }
        try indices.withUnsafeBytes { try Data($0).write(to: URL(fileURLWithPath: out + ".indices")) }
        // Stations, to aim the camera.
        let stations = world.stations.map { s -> [String: Any] in
            let p = scene.point(worldX: Double(s.point.x), worldY: Double(s.point.y), height: 0)
            return ["name": s.name, "x": p.x, "z": p.z]
        }
        let cell = Double(Land.cellLength)
        let low = CitySkyline(world: world).lots(in: scene.region).filter { ($0.use == .residential || $0.use == .commercial) && $0.density <= 2 }
            .map { lot -> [String: Any] in
                let p = scene.point(worldX: (Double(lot.column) + 0.5) * cell, worldY: (Double(lot.row) + 0.5) * cell, height: 0)
                return ["x": p.x, "z": p.z, "d": lot.density, "shop": lot.use == .commercial]
            }
        // Uniforms as Map3DMetalView.uniformValues writes them, for a few
        // cameras at MAP3D_MINUTE of the day (14:00 unless set).
        let lighting = Map3DLighting(minute: ProcessInfo.processInfo.environment["MAP3D_MINUTE"].flatMap(Double.init) ?? 14 * 60)
        func rgb(_ c: PopTravel.RGB) -> (Float, Float, Float) { (Float(c.red) / 255, Float(c.green) / 255, Float(c.blue) / 255) }
        func padded(_ light: SIMD3<Float>) -> [Float] { [light.x, light.y, light.z, 0] }
        func uniforms(_ camera: Map3DCamera) -> [Float] {
            let sky = rgb(lighting.sky)
            let eye = camera.eye, far = camera.depthRange.far
            return camera.viewProjection.columns
                + [lighting.sunDirection.x, lighting.sunDirection.y, lighting.sunDirection.z, 0]
                + padded(lighting.sunLight) + padded(lighting.skyLight) + padded(lighting.groundLight)
                + [sky.0, sky.1, sky.2, 0] + [eye.x, eye.y, eye.z, 1] + [camera.distance * 1.6, far * 0.9, 0, 0]
                + [Map3DShadow.strength(lighting), lighting.lights, 0, 0] + Map3DLighting.nightWindowRows
        }
        let presets: [(String, Map3DPoint, Float, Float)] = (ProcessInfo.processInfo.environment["MAP3D_VIEWS"] ?? "close:0:0:140:35;mid:0:0:420:-25;far:0:0:1300:0")
            .split(separator: ";").map { part in
                let f = part.split(separator: ":").map(String.init)
                return (f[0], Map3DPoint(Float(f[1])!, 0, Float(f[2])!), Float(f[3])!, Float(f[4])! * .pi / 180)
            }
        var views: [String: [Float]] = [:]
        // What the app would draw from each camera (decision 238): the
        // chunks in view, each in full or simplified.
        var drawn: [String: Int] = [:], drawnInFull: [String: Int] = [:]
        for (name, target, distance, heading) in presets {
            let camera = Map3DCamera(target: target, distance: distance, heading: heading, viewport: (1600, 900))
            views[name] = uniforms(camera)
            drawnInFull[name] = scene.chunks.filter { Map3DCamera.sees(minimum: $0.minimum, maximum: $0.maximum, through: camera.viewProjection) }
                .reduce(0) { $0 + $1.mesh.triangleCount }
            drawn[name] = scene.chunks.filter { Map3DCamera.sees(minimum: $0.minimum, maximum: $0.maximum, through: camera.viewProjection) }
                .reduce(0) { total, chunk in
                    total + (camera.wantsDetail(minimum: chunk.minimum, maximum: chunk.maximum)
                        ? chunk.mesh.triangleCount : chunk.baseIndexCount / 3 + chunk.far.triangleCount)
                }
        }
        let uses = CitySkyline(world: world).lots(in: scene.region).filter { !$0.isOpenGround }
            .map { lot -> [String: Any] in
                let p = scene.point(worldX: (Double(lot.column) + 0.5) * cell, worldY: (Double(lot.row) + 0.5) * cell, height: 0)
                return ["x": p.x, "z": p.z, "d": lot.density, "use": lot.use.rawValue]
            }
        try JSONSerialization.data(withJSONObject: ["drawn": drawn, "drawnInFull": drawnInFull, "uses": uses, "stations": stations, "low": Array(low.prefix(400)), "triangles": indices.count / 3, "views": views, "sky": [rgb(lighting.sky).0, rgb(lighting.sky).1, rgb(lighting.sky).2]])
            .write(to: URL(fileURLWithPath: out + ".json"))
    }
}
